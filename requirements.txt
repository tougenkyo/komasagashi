# komasagashi.py の依存ライブラリ
#
# Windows なら setup.bat をダブルクリックすれば、下記を含めて自動で入る。
# 手動で入れる場合は、先に torch と torchvision を入れてからこのファイルを使うこと:
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements.txt
# （PyPI の Windows 版 torch は CPU 専用のため。GPU が無ければ末尾を /whl/cpu にする）
#
# GUI の tkinter と索引の sqlite3 は Windows 版 Python に標準同梱のため記載しない。

# OCR エンジン本体（comic-text-detector + manga-ocr-base）。
# 新しい版は未確認のため、動作確認した版に固定している。
mokuro==0.1.3
manga-ocr==0.1.14
# manga-ocr が使う。5 系は API が大きく変わったため未確認。
transformers<5

# コマ検出に使用（mokuro も使う）。任意機能の PaddleOCR が入れる opencv-contrib-python と
# 同じ版にそろえる（どちらも同じ cv2 フォルダに入るため、版が違うと壊れることがある）。
opencv-python==4.10.0.84

# 画像の読み込み・リサイズと、検索タブのプレビュー表示（ImageTk）に使用。
# manga-ocr が 10.0 以上を要求する。
Pillow>=10.0

# mokuro の推論に使用。torch と torchvision は必ず同じ配布元（CUDA 版なら同じ index-url）から
# 入れること。torchvision だけ PyPI から入ると、torch が CPU 版に置き換えられることがある。
torch
torchvision

# 書庫の中の画像を読む（zip / cbz は Python 標準の zipfile で読む）。
# 7z / cb7。py7zr 1.1.3 で動作確認。
py7zr>=1.1,<2
# rar / cbr。一覧は rarfile が読み、展開は PC にある WinRAR（UnRAR）・7-Zip・Windows 標準の tar を使う。
rarfile>=4.2,<5

# comic-text-detector は numpy 2 で削除された np.bool8 などを使うため、numpy 2 では起動しない。
# 1.24 以上は任意機能の PaddleOCR の要求（1.24 で消えた np.int はアプリ側で補っている）。
numpy>=1.24,<2

# mokuro が依存する comic-text-detector が pkg_resources を import するが、
# setuptools 81 以降では pkg_resources が削除されており起動時に ImportError になる。
setuptools<81
